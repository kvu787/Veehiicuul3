function(compile_shader source name entry profile)
    set(header "${GeneratedDirectory}/${name}.h")
    add_custom_command(OUTPUT "${header}"
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${GeneratedDirectory}"
        COMMAND "${DXC_EXECUTABLE}" -E ${entry} -T ${profile} -O3 -Ges -WX -all_resources_bound
            -D SIMPLE_PAINT_MATERIAL_COUNT=32
            -Qstrip_debug -Qstrip_reflect -Fh "${header}" -Vn "${name}" "${source}"
        DEPENDS "${source}" ${ARGN} VERBATIM)
    set(ShaderHeaders ${ShaderHeaders} "${header}" PARENT_SCOPE)
endfunction()
